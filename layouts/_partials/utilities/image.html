Ahora usa el parámetro `alt` (vacío por defecto, válido para imágenes decorativas). */}} {{ if not .src }} {{ errorf "Image Optimization: the partial 'utilities/image' is missing the 'src' argument." }} {{ end }} {{ $resource := dict }} {{ with page.Resources.Get .src }}{{ $resource = . }}{{ end }} {{ with resources.Get .src }}{{ $resource = . }}{{ end }} {{ if not $resource }}{{ errorf "Image Optimization: the image '%s' has not been found." .src }}{{ end }} {{ $alt := .alt | default "" }} {{ $breakpoints := (slice 576 768 992 1200 1400) }} {{/* Genera las variantes WebP solo si el recurso es raster (los SVG no tienen ancho). */}} {{ $srcset := slice }} {{ if gt $resource.Width 0 }} {{ range $breakpoints }} {{ if ge $resource.Width . }} {{ $image := $resource.Resize (printf "%dx webp" .) }} {{ $srcset = $srcset | append (slice (printf "%s %dw" $image.Permalink $image.Width)) }} {{ end }} {{ end }} {{ end }}
